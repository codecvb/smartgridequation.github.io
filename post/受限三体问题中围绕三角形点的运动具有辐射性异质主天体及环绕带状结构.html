摘要：本文通过Python实现了圆形限制性三体问题(CR3BP)的数值模拟，研究了三个双星系统(Archid、Xi Bootis和Kruger 60)的三角拉格朗日点(L4和L5)附近的轨道动力学。程序使用SciPy求解微分方程，计算了等边三角形构型的拉格朗日点位置，并在这些点附近施加微小扰动模拟轨道运动。结果显示，所有三个双星系统的三角点附近都存在周期性轨道，验证了拉格朗日点在限制性三体问题中的稳定性特性。研究采用无量纲化处理，可视化呈现了主星、次星、三角点及附近轨道的空间分布关系。

Python代码

import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import solve_ivp

# ---------------- 物理与数学参数 ------------------
# 论文提供的三双星系统质量比mu，部分参数简化
binaries = {
    "Archid": {"mu": 0.3942, "name": "Archid"},
    "Xi_Bootis": {"mu": 0.49, "name": "Xi Bootis"},
    "Kruger_60": {"mu": 0.4433, "name": "Kruger 60"},
}

def cr3bp_eq(t, Y, mu):
    """
    CR3BP标准模型的微分方程
    输入：
      t: 时间（无量纲）
      Y: 状态向量[x, y, vx, vy]
      mu: 质量比
    输出：
      [vx, vy, ax, ay]
    """
    x, y, vx, vy = Y
    r1 = np.sqrt((x + mu)**2 + y**2)
    r2 = np.sqrt((x - 1 + mu)**2 + y**2)

    ax = 2 * vy + x - (1 - mu)*(x + mu)/r1**3 - mu*(x - 1 + mu)/r2**3
    ay = -2 * vx + y - (1 - mu)*y/r1**3 - mu*y/r2**3
    return [vx, vy, ax, ay]

def lagrange_triangular_points(mu):
    """
    计算R3BP的两个三角拉格朗日点L4和L5坐标
    L4和L5是等边三角形顶点，坐标为(0.5 - mu, ± sqrt(3)/2)
    """
    xL = 0.5 - mu
    yL = np.sqrt(3) / 2
    return (xL, yL), (xL, -yL)

def simulate_orbit_around_point(mu, point, duration=10, dt=0.01):
    """
    在三角点附近模拟短暂轨迹，考察附近运动趋势
    起始条件为三角点坐标加上微小扰动
    """
    t_eval = np.arange(0, duration, dt)
    # 在原点加扰动速度，模拟环绕运动
    v_perturb = 0.05
    y0 = [point[0], point[1], 0, v_perturb]
    sol = solve_ivp(cr3bp_eq, [0, duration], y0, args=(mu,), t_eval=t_eval, rtol=1e-9, atol=1e-9)
    return sol

# ---------------- 主程序 ------------------

plt.figure(figsize=(18, 5))

for i, (key, param) in enumerate(binaries.items(), 1):
    mu = param["mu"]
    name = param["name"]

    # 获取三角点
    L4, L5 = lagrange_triangular_points(mu)

    # 轨迹模拟
    sol_L4 = simulate_orbit_around_point(mu, L4)
    sol_L5 = simulate_orbit_around_point(mu, L5)

    # 画板设置
    ax = plt.subplot(1,3,i)

    # 质量主体位置（主星在-x=mu）
    ax.scatter(-mu, 0, color='blue', label="Primary (mass 1-$\mu$)", s=100)
    ax.scatter(1-mu, 0, color='red', label=f"Secondary ($\mu$)", s=50)

    # 三角点位置
    ax.scatter([L4[0], L5[0]], [L4[1], L5[1]], color='green', s=80, label='Triangular Points L4, L5')

    # 模拟轨迹
    ax.plot(sol_L4.y[0], sol_L4.y[1], 'g--', linewidth=1.5, label='Orbit around L4')
    ax.plot(sol_L5.y[0], sol_L5.y[1], 'g-', linewidth=1.5, label='Orbit around L5')

    ax.set_title(f"{name} System\n$\mu$ = {mu:.4f}", fontsize=14)
    ax.set_xlabel('x (normalized)')
    ax.set_ylabel('y (normalized)')
    ax.legend(loc='upper right', fontsize=8)
    ax.set_aspect('equal')
    ax.grid(True)
    ax.set_xlim([-1.2,1.2])
    ax.set_ylim([-1.2,1.2])

plt.suptitle("Restricted Three Body Problem: Triangular Equilibrium Points and Local Orbits", fontsize=16)
plt.tight_layout(rect=[0,0,1,0.95])
plt.show()

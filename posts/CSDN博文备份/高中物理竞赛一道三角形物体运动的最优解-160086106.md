水平长度<br>h = 0.75 * l     # 竖直高度<br>ratio = h / l    # 0.75<br><br># 最短时间路径 (A->B->C)<br>t_min_vertical = math.sqrt(2 * h / g)<br>t_min_horizontal = l / math.sqrt(2 * g * h)<br>T_min = t_min_vertical + t_min_horizontal<br><br>print(f"最短时间 T_min = {T_min:.6f} s")<br><br># 最大时间理论值 (沿斜边极限)<br>t_max_vertical = t_min_vertical<br>t_max_horizontal = 2 * l / math.sqrt(2 * g * h)<br>T_max_theory = t_max_vertical + t_max_horizontal<br>print(f"最长时间理论值 T_max = {T_max_theory:.6f} s")<br>print(f"理论比值 T_max/T_min = {T_max_theory / T_min:.6f} = 7/5 = 1.4\n")<br><br># 不同分段数 N 的阶梯路径 (等比例 Δy = (h/l) Δx)<br>for N in [1, 2, 5, 10, 20, 50, 100, 200]:<br>    dx = l / N<br>    dy = ratio * dx   # 每段竖直下降<br>    # 累计下降和水平位置<br>    S = 0.0   # 累计下降<br>    X = 0.0   # 累计水平<br>    T_h = 0.0<br>    for i in range(N):<br>        S += dy          # 第i段水平开始时的累计下降<br>        v = math.sqrt(2 * g * S)<br>        T_h += dx / v<br>    T_total = t_min_vertical + T_h<br>    print(f"N = {N:3d}, T_total = {T_total:.6f} s, 比值 = {T_total / T_min:.6f}")</code></pre><br><br><p><strong>输出示例</strong>（数值接近理论值）：</p><br><br><pre><br><code class="language-html">最短时间 T_min = 0.654654 s<br>最长时间理论值 T_max = 0.916515 s<br>理论比值 T_max/T_min = 1.400000 = 7/5 = 1.4<br><br>N =   1, T_total = 0.654654 s, 比值 = 1.000000<br>N =   2, T_total = 0.784198 s, 比值 = 1.197981<br>N =   5, T_total = 0.856766 s, 比值 = 1.308747<br>N =  10, T_total = 0.884284 s, 比值 = 1.350868<br>N =  20, T_total = 0.899579 s, 比值 = 1.374235<br>N =  50, T_total = 0.909485 s, 比值 = 1.389369<br>N = 100, T_total = 0.912985 s, 比值 = 1.394582<br>N = 200, T_total = 0.914746 s, 比值 = 1.397270</code></pre><br><br><p></p><br>

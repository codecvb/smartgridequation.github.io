body_scale; % 眼睛半径<br>eye_offset_x = 1.0 * body_scale; % 眼睛在x方向上的偏移<br>eye_offset_y = 0.5 * body_scale; % 眼睛在y方向上的偏移<br><br>% 鼻子<br>nose_width = 0.6 * body_scale; % 鼻子宽度<br>nose_height = 0.4 * body_scale; % 鼻子高度<br><br>% 嘴巴<br>mouth_width = 1.2 * body_scale; % 嘴巴宽度<br>mouth_height = 0.5 * body_scale; % 嘴巴高度<br><br>% 身体<br>body_width = 3.5 * body_scale; % 身体宽度<br>body_height = 3.0 * 绘制眼睛<br>% 左眼<br>rectangle('Position', [head_center(1)-eye_offset_x-eye_radius, head_center(2)+eye_offset_y-eye_radius, ...<br>                       2*eye_radius, 2*eye_radius], ...<br> 左手臂<br>rectangle('Position', [-body_width/2-limb_width, body_height/2, limb_width, limb_height], ...<br>          'Curvature', [0.3, 0.3], 'FaceColor', bear_color, 'EdgeColor', 'none');<br>%

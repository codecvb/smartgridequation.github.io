<pre><code class="language-python">import cv2<br>import pywt<br>import numpy as np<br><br><br>def wavelet_denoise(image, wavelet='db4', level=1, mode='soft'):<br>    # 将图像转换为灰度图<br>    if len(image.shape) == 3:<br>        image = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)<br><br>    # 进行小波分解<br>    coeffs = pywt.wavedec2(image, wavelet, level=level)<br><br>    # 对每个细节系数应用阈值处理<br>    threshold = np.std(coeffs[-1]) * np.sqrt(2 * np.log2(image.size))<br>    new_coeffs = [coeffs[0]]<br>    for detail_coeffs in coeffs[1:]:<br>        new_detail_coeffs = [pywt.threshold(d, threshold, mode=mode) for d in detail_coeffs]<br>        new_coeffs.append(new_detail_coeffs)<br><br>    # 进行小波重构<br>    denoised_image = pywt.waverec2(new_coeffs, wavelet)<br><br>    # 将像素值限制在 0 到 255 之间<br>    denoised_image = np.clip(denoised_image, 0, 255).astype(np.uint8)<br><br>    return denoised_image<br><br><br># 读取图像<br>image = cv2.imread('2.png')<br><br># 进行小波去噪<br>denoised_image = wavelet_denoise(image)<br><br>cv2.imwrite('3.png', denoised_image)<br><br># 显示原始图像和去噪后的图像<br>cv2.imshow('Original Image', image)<br>cv2.imshow('Denoised Image', denoised_image)<br>cv2.waitKey(0)<br>cv2.destroyAllWindows()<br>    </code></pre> <br><p style="text-align:center"><img alt="" src="https://i-blog.csdnimg.cn/direct/d06c20b57c5942a98428f7799a97ae2f.png" /></p>

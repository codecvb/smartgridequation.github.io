<p>本文介绍了一个使用Python Matplotlib库创建的月相变化动画程序。该程序通过绘制地球、月球轨道和太阳光照，模拟展示了月球绕地球公转时不同位置呈现的月相变化（包括新月、上弦月、满月等下弦月等）。动画采用红色月球配合白色受光面的视觉效果，准确表现了月相变化的原理。程序设置了120帧动画循环，每帧计算月球轨道角度和相应月相名称，最终输出为GIF格式的月相变化演示动画。该实现结合了天文学知识和数据可视化技术，为理解月相变化提供了直观的教学工具。</p><br><br><p style="text-align:center"><img alt="" src="https://i-blog.csdnimg.cn/direct/ac5eabca6dd14f379a3107eacd5c60b4.jpeg" /></p><br><br><p class="img-center"><img alt="" src="https://i-blog.csdnimg.cn/direct/440ea377ee3a4b688dac34e2629a6309.gif" /></p><br><br><pre><br><code class="language-python">import numpy as np<br>import matplotlib.pyplot as plt<br>from matplotlib.patches import Circle<br>from matplotlib.animation import FuncAnimation, PillowWriter<br>from matplotlib import rcParams<br><br># 中文字体<br>rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei", "Arial Unicode MS"]<br>rcParams["axes.unicode_minus"] = False<br><br># ----------------------------<br># 参数设置<br># ----------------------------<br>fig, ax = plt.subplots(figsize=(8, 8))<br>ax.set_aspect("equal")<br>ax.set_xlim(-4.2, 4.8)<br>ax.set_ylim(-4.2, 4.2)<br>ax.axis("off")<br><br>orbit_r = 2.8<br>earth_r = 0.5<br>moon_r = 0.28<br>sun_x, sun_y = 4.2, 3.0<br>frames = 120<br><br># ----------------------------<br># 绘制“红色月球 + 白色投影”<br># ----------------------------<br>def draw_moon_red_white(ax, x, y, r, phase_deg):<br>    # 红色底盘<br>    ax.add_patch(Circle((x, y), r, color="#CC0000", ec="black", lw=1.2))<br><br>    # 新月<br>    if phase_deg == 0:<br>        return<br><br>    # 满月：整颗变白<br>    if phase_deg == 180:<br>        ax.add_patch(Circle((x, y), r, color="white", ec="black", lw=1.2))<br>        return<br><br>    # 白色“投影/受光面”<br>    phi = np.deg2rad(phase_deg)<br><br>    if phase_deg < 180:<br>        # 盈月：右侧白<br>        ax.add_patch(Circle((x, y), r, color="white", ec="none"))<br>        shift = r * np.cos(phi)<br>        ax.add_patch(Circle((x - shift, y), r, color="#CC0000", ec="none"))<br>    else:<br>        # 亏月：左侧白<br>        ax.add_patch(Circle((x, y), r, color="white", ec="none"))<br>        shift = r * np.cos(np.pi - phi)<br>        ax.add_patch(Circle((x + shift, y), r, color="#CC0000", ec="none"))<br><br>    # 外轮廓<br>    ax.add_patch(Circle((x, y), r, fill=False, ec="black", lw=1.2))<br><br># ----------------------------<br># 月相文字<br># ----------------------------<br>def get_phase_name(angle_deg):<br>    if angle_deg < 22.5 or angle_deg >= 337.5:<br>        return "新月"<br>    elif angle_deg < 67.5:<br>        return "蛾眉月"<br>    elif angle_deg < 112.5:<br>        return "上弦月"<br>    elif angle_deg < 157.5:<br>        return "盈凸月"<br>    elif angle_deg < 202.5:<br>        return "满月"<br>    elif angle_deg < 247.5:<br>        return "亏凸月"<br>    elif angle_deg < 292.5:<br>        return "下弦月"<br>    else:<br>        return "亏眉月"<br><br># ----------------------------<br># 动画更新函数<br># ----------------------------<br>def update(frame):<br>    ax.clear()<br>    ax.set_aspect("equal")<br>    ax.set_xlim(-4.2, 4.8)<br>    ax.set_ylim(-4.2, 4.2)<br>    ax.axis("off")<br>    ax.set_facecolor("white")<br><br>    # 当前月球轨道角度<br>    angle_deg = frame * 360 / frames<br>    theta = np.deg2rad(angle_deg)<br><br>    # 月球位置<br>    moon_x = orbit_r * np.cos(theta)<br>    moon_y = orbit_r * np.sin(theta)<br><br>    # 月相角：0°新月，180°满月<br>    phase_deg = angle_deg if angle_deg <= 180 else 360 - angle_deg<br><br>    # 太阳光<br>    ax.annotate(<br>        "",<br>        xy=(-3.0, 3.0),<br>        xytext=(4.0, 3.0),<br>        arrowprops=dict(arrowstyle="->", lw=2.5, color="orange")<br>    )<br>    ax.text(4.05, 3.15, "太阳光", color="orange", fontsize=12, ha="right")<br><br>    # 太阳<br>    sun = Circle((sun_x, sun_y), 0.35, color="gold", ec="black", lw=1.2)<br>    ax.add_patch(sun)<br>    ax.text(sun_x, sun_y, "太阳", ha="center", va="center", fontsize=11, weight="bold")<br><br>    # 地球<br>    earth = Circle((0, 0), earth_r, color="#2E86DE", ec="black", lw=1.5)<br>    ax.add_patch(earth)<br>    ax.text(0, 0, "地球", ha="center", va="center", color="white", fontsize=13, weight="bold")<br><br>    # 轨道<br>    orbit = Circle((0, 0), orbit_r, fill=False, ls="--", ec="gray", lw=1.2)<br>    ax.add_patch(orbit)<br><br>    # 地月连线<br>    ax.plot([0, moon_x], [0, moon_y], color="gray", lw=1, ls=":")<br><br>    # 月球<br>    draw_moon_red_white(ax, moon_x, moon_y, moon_r, phase_deg)<br>    ax.text(moon_x, moon_y - 0.45, "月球", ha="center", va="top", fontsize=11)<br><br>    # 文字说明<br>    phase_text = get_phase_name(angle_deg)<br>    ax.text(-4.0, 3.6, f"月相动画：{phase_text}", fontsize=15, weight="bold")<br>    ax.text(2.1, -3.7, f"月球轨道角度：{int(angle_deg)}°", fontsize=11)<br><br>    # 轨道方向箭头<br>    ax.annotate(<br>        "",<br>        xy=(orbit_r * np.cos(np.deg2rad(70)), orbit_r * np.sin(np.deg2rad(70))),<br>        xytext=(orbit_r * np.cos(np.deg2rad(30)), orbit_r * np.sin(np.deg2rad(30))),<br>        arrowprops=dict(arrowstyle="->", lw=1.8, color="black")<br>    )<br><br># ----------------------------<br># 生成 GIF<br># ----------------------------<br>anim = FuncAnimation(fig, update, frames=frames, interval=80)<br>gif_name = "moon_phase_red_white.gif"<br>anim.save(gif_name, writer=PillowWriter(fps=12))<br>print(f"已生成动画文件：{gif_name}")<br>plt.close(fig)</code></pre><br><br><p></p><br>

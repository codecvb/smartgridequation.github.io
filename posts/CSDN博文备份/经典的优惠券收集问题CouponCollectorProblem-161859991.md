<p><strong>题目描述：</strong>一共有 n 种不同的优惠券（Coupons），每次抽取时，获得任意一种优惠券的概率均等（均为 1/n），且每次抽取相互独立。求：为了集齐所有 n 种优惠券，需要抽取次数的数学期望，并以带分数的形式输出结果。</p><br><br><p><strong>数据范围：</strong> 1≤n≤33。</p><br><br><p><img alt="" height="542" src="https://i-blog.csdnimg.cn/direct/f1fbf190de174c12b981f50ad578a145.png" width="660" /></p><br><br><p><img alt="" height="353" src="https://i-blog.csdnimg.cn/direct/06e35a728a8547178edf599dd05d47ab.png" width="655" /></p><br><br><pre><br><code class="language-cpp">#include <iostream><br>#include <algorithm><br>using namespace std;<br><br>typedef long long ll;<br><br>ll gcd(ll a, ll b) {<br>    return b == 0 ? a : gcd(b, a % b);<br>}<br><br>int main() {<br>    int n;<br>    cin >> n;<br><br>    ll numerator = 0, denominator = 1;<br>    // 计算调和数 H_n 的分数形式<br>    for (int i = 1; i <= n; ++i) {<br>        ll lcm = denominator / gcd(denominator, (ll)i) * i;<br>        numerator = numerator * (lcm / denominator) + (lcm / i);<br>        denominator = lcm;<br>        ll g = gcd(numerator, denominator);<br>        numerator /= g;<br>        denominator /= g;<br>    }<br><br>    // 乘以 n，再约分<br>    numerator *= n;<br>    ll g = gcd(numerator, denominator);<br>    numerator /= g;<br>    denominator /= g;<br><br>    // 输出带分数形式<br>    if (denominator == 1) {<br>        cout << numerator << endl;<br>    } else {<br>        ll integer = numerator / denominator;<br>        ll remainder = numerator % denominator;<br>        if (integer > 0) {<br>            cout << integer << " " << remainder << "/" << denominator << endl;<br>        } else {<br>            cout << remainder << "/" << denominator << endl;<br>        }<br>    }<br><br>    return 0;<br>}</code></pre><br><br><p><img alt="" height="597" src="https://i-blog.csdnimg.cn/direct/ba1cf1490e9d4813ad42decc2f877fc0.png" width="665" /></p><br><br><p><img alt="" height="571" src="https://i-blog.csdnimg.cn/direct/6e02b9ff353f47feb67a3f6af711d789.png" width="656" /></p><br><br><p><img alt="" height="309" src="https://i-blog.csdnimg.cn/direct/b62fa813f5ab4762b2fc694c7e8a132d.png" width="663" /></p><br><br><p><img alt="" height="422" src="https://i-blog.csdnimg.cn/direct/ffc1a233763c421986444f06e056c522.png" width="800" /></p><br><br><p><img alt="" height="308" src="https://i-blog.csdnimg.cn/direct/f79921a91eef41e3ab49811282b6878d.png" width="809" /></p><br>

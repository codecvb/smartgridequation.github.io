% 万有引力常数(N·m²/kg²)<br>    AU = 1.496e11;   % 天文单位(m)<br>    day = 86400;     % 每日秒数<br>    <br>    % 太阳质量(kg)<br>    sun_mass = 1.989e30;<br>    <br>    % 行星数据 [质量(kg) 轨道半长轴(AU) 离心率 轨道倾角(deg) 颜色RGB]<br>    planets = {<br>        '水星', 3.3011e23, 0.387, 0.206, 7.0, [0.7 0.7 0.7];<br>        '金星', 4.8675e24, 0.723, 0.007, 3.4, [0.9 0.6 0.1];<br>        '地球', 5.9724e24, 1.000, 0.017, 0.0, [0.1 0.5 0.8];<br>        '火星', 6.4171e23, 1.524, 0.093, 1.9, [0.8 0.3 0.1];<br>        '木星', 1.8982e27, 5.203, 0.048, 1.3, [0.8 0.7 0.5];<br>        '土星', 5.6834e26, 9.537, 0.056, 2.5, [0.9 0.8 0.6];<br>        '天王星', 8.6810e25, 19.191, 0.046, 0.8, [0.6 0.8 0.9];<br>        '海王星', 1.0243e26, 30.069, 0.010, 1.8, [0.3 0.5 0.9];<br>        '冥王星', 1.303e22, 39.482, 0.248, 17.2, [0.7 0.5 0.5]<br>    };<br>    <br>    % 创建3D图形窗口<br>    figure('Color','k','Position',[100 100 1200 800])<br>    hold on<br>    axis equal<br>    grid on<br>    view(3)<br>    xlabel('X (AU)'), ylabel('Y (AU)'), zlabel('Z (AU)')<br>    title('太阳系行星轨道模拟','Color','w')<br>    set(gca,'Color','k','XColor','w','YColor','w','ZColor','w')<br>    <br>    % 绘制太阳<br>    [X,Y,Z] = sphere(10);<br>    surf(X*0.2,Y*0.2,Z*0.2,'FaceColor','y','EdgeColor','none')<br>    light('Position',[0 0 0],'Style','local','Color','y')<br>    <br>    % 计算并绘制各行星轨道<br>    theta = linspace(0,2*pi,500);<br>    for i = 1:size(planets,1)<br>        a = planets{i,3}*AU;       % 半长轴(m)<br>        e = planets{i,4};          % 离心率<br>        r = a*(1-e)./(1+e*cos(theta)); % 极坐标轨道方程<br>        <br>        % 转换为笛卡尔坐标并考虑轨道倾角<br>        inclination = deg2rad(planets{i,5});<br>        x = r.*cos(theta);<br>        y = r.*sin(theta)*cos(inclination);<br>        z = r.*sin(theta)*sin(inclination);<br>        <br>        % 绘制轨道线<br>        plot3(x/AU,y/AU,z/AU,'Color',planets{i,6},'LineWidth',1);<br>        <br>        % 计算轨道周期(秒) - 开普勒第三定律<br>        T = sqrt(4*pi*a/(G*(sun_mass+planets{i,2})));<br>        <br>        % 创建行星动画对象<br>        planet_plot(i) = plot3(NaN,NaN,NaN,'o','Color',planets{i,6},...<br>            'MarkerSize',8,'MarkerFaceColor',planets{i,6});<br>        text(a/AU,0,0,planets{i,1},'Color','w')<br>    end<br>    <br>    % 动画参数设置<br>    days_per_frame = 5;    % 每帧天数<br>    total_days = 100;     % 总模拟天数<br>    frames = total_days/days_per_frame;<br>    <br>    % --- 新增部分：GIF保存准备 ---<br>    % 定义GIF文件名<br>    gif_filename = 'solar_system_simulation.gif';<br>    % --- 新增结束 ---<br>    <br>    % 动画循环<br>    for f = 1:frames<br>        time = f*days_per_frame*day;<br>        <br>        for i = 1:size(planets,1)<br>            a = planets{i,3}*AU;<br>            e = planets{i,4};<br>            <br>            % 计算当前角度(简化模型)<br>            T = sqrt(4*pi*a/(G*(sun_mass+planets{i,2})));<br>            angle = 2*pi*mod(time/T,1);<br>            <br>            % 计算当前位置<br>            r = a*(1-e)/(1+e*cos(angle));<br>            inclination = deg2rad(planets{i,5});<br>            x = r*cos(angle);<br>            y = r*sin(angle)*cos(inclination);<br>            z = r*sin(angle)*sin(inclination);<br>            <br>            % 更新行星位置<br>            set(planet_plot(i),'XData',x/AU,'YData',y/AU,'ZData',z/AU)<br>        end<br>        <br>        drawnow<br>        <br>        % --- 新增部分：保存当前帧到GIF ---<br>        % 1. 获取当前坐标轴的图像数据<br>        frame = getframe(gca);<br>        % 2. 将图像数据转换为索引图像，这是GIF格式所要求的<br>        %    'colormap'参数确保颜色准确<br>        im = frame2im(frame);<br>        [imind, cm] = rgb2ind(im, 256);<br>        <br>        % 3. 写入GIF文件<br>        %    对于第一帧，使用'w'模式创建文件<br>        %    对于后续帧，使用'a'模式追加到文件末尾<br>        %    'DelayTime'设置每帧的显示时间(秒)<br>        %    'LoopCount',inf 使GIF无限循环播放<br>        if f == 1<br>            imwrite(imind, cm, gif_filename, 'GIF', 'LoopCount', inf, 'DelayTime', 0.05);<br>        else<br>            imwrite(imind, cm, gif_filename, 'GIF', 'WriteMode', 'append', 'DelayTime', 0.05);<br>        end<br>        % --- 新增结束 ---<br>        <br>        pause(0.05)<br>    end<br>    <br>    % --- 新增部分：提示信息 ---<br>    disp(['动画已成功保存为 GIF 文件: ', fullfile(pwd, gif_filename)]);<br>    % --- 新增结束 ---<br>end</code></pre><br><br><p></p><br>

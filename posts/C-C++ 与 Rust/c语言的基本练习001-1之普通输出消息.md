分别输出100以前的自然数的倒数和：


```cpp
#include<math.h>
#include<stdio.h>

int main(){

int i = 100;

float sum = 0.0;

for(i = 1;i<=100;i++){

    printf("第%d个倒数累加和是：%.4f;\n",i,sum=sum+1/((float)i));

}

}
```

```cs
using System;
using System.Text;
class Hello
{
    static void Main()
    {
        int i = 100;
        float sum = 0.0f;
        for(i=0;i<=100;i++){
            sum = sum + 1/(i+1.0f);
            Console.Write("the reciprocal sum of nature numbers before ");
            Console.WriteLine(i+1);
            Console.Write("is:");
            Console.WriteLine(sum.ToString("0.0000"));
        }
    }
}
```
